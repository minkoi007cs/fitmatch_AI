"""Dry-run only: synthetic raw OCR + diagnostic copies follow source binding; models/derived protected."""
import hashlib
import json
from pathlib import Path
import sys
import tempfile
import uuid

ROOT = Path(__file__).resolve().parent
sys.path.insert(0,str(ROOT.parent/'quality03'))
from retention import cleanup_plan

report = ROOT.parents[1]/'reports/ai/qint_p'
with tempfile.TemporaryDirectory(prefix='ret-', dir=ROOT) as temporary:
    root = Path(temporary)
    contents = [('source', b'public synthetic image placeholder', '.jpg'),
                ('source-evidence',b'{"raw_ocr":"UNVERIFIED_SYNTHETIC"}', '.json'),
                ('source-evidence',b'{"phase":"synthetic diagnostic copy"}\n', '.jsonl')]
    binding = {key:str(uuid.UUID(int=i)) for i,key in enumerate(['owner_id','job_id','source_revision'],1)}
    binding['image_sha256'] = hashlib.sha256(contents[0][1]).hexdigest()
    prefix = f"{uuid.UUID(binding['job_id']).hex}/{uuid.UUID(binding['source_revision']).hex}"
    entries = []
    for kind,data,suffix in contents:
        digest = hashlib.sha256(data).hexdigest()
        relative = f'{prefix}/{kind}/{digest}{suffix}'
        path = root/relative; path.parent.mkdir(parents=True,exist_ok=True); path.write_bytes(data)
        entries.append({'relative_path':relative,'sha256':digest})
    (root/prefix/'binding.json').write_text(json.dumps({'version':'pc-ai.job-source-binding.v1','binding':binding}))
    params = dict(binding=binding,tombstone={**binding,'deleted':True,'job_attempt':2},
                  entries=entries,active_processes=0,job_attempt=2)
    plan = cleanup_plan(root,**params)
    assert len(plan['candidates']) == 3 and not plan['executed']
    assert all((root/item['relative_path']).is_file() for item in entries)
    protected = []
    for wrong in ['models/qwen3-vl-4b/model.safetensors', entries[0]['relative_path'].replace('/source/','/derived/')]:
        try: cleanup_plan(root,**{**params,'entries':[{'relative_path':wrong,'sha256':binding['image_sha256']}]})
        except ValueError: protected.append(wrong.split('/')[0] if wrong.startswith('models') else 'derived')
        else: raise AssertionError('Protected artifact was included')
    try: cleanup_plan(root,**{**params,'active_processes':1})
    except ValueError: active_refused = True
    else: raise AssertionError('Undrained process accepted')
    output = {'status':'dry_run_passed','scope':'synthetic bound source, raw OCR and diagnostic copies',
        'candidate_count':3,'existing_files_unchanged_before_fixture_teardown':True,
        'protected_refused':protected,'active_process_refused':active_refused,
        'executed':False,'physical_source_deletions':0,
        'actual_cli_outputs':'local public-reference diagnostics retained; no deployed binding collector or deletion ledger',
        'personal_input_cleanup':'Not run; personal inputs absent'}
(report/'retention-dryrun.json').write_text(json.dumps(output,indent=2)+'\n')
print(json.dumps(output))
