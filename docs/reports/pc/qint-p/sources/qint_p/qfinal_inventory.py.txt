def load_backend_registry():
    raw=(ROOT/'backend-environments.snapshot.json').read_bytes()
    pin=json.loads((ROOT/'backend-environments.pin.json').read_text())
    registry=json.loads(raw)
    if hashlib.sha256(raw).hexdigest()!=pin['sha256']:
        raise RuntimeError('Backend registry snapshot hash mismatch; no fallback')
    canonical=registry['canonical']
    parsed=urlsplit(canonical['apiUrl'])
    if (registry.get('version')!=1 or registry.get('decision')!='BACKEND-024' or canonical.get('projectName')!='fitmatch-ai'
        or canonical.get('projectRef')!='gohczmqykjkrgdblgbog'

def configured(value):
    if not isinstance(value,str) or not value.strip():return False
    text=value.strip().strip('\"\'')
    if re.search(r'(?i)(replace|placeholder|your[-_ ]|required|example|change[-_]?me|not[-_]a[-_]credential)',text):return False
    return not (text.startswith('<') or (text.startswith('%') and text.endswith('%')))

def endpoint(value):
    if not configured(value):return False
    try:
        parsed=urlsplit(value)
        return (parsed.scheme=='https' and bool(parsed.hostname) and parsed.username is None
                and not parsed.hostname.endswith(('.invalid','.example','.supabase.co')) and parsed.hostname not in {'localhost','127.0.0.1'})
    except ValueError:return False

def summary(alias,values,*,kind='environment'):
    urls=ENDPOINT_KEYS if kind=='environment' else {'appUrl','testAppUrl','testEndpoint'}
    tokens=TOKEN_KEYS if kind=='environment' else {'workerToken','testWorkerToken','token'}
    labels=LABEL_KEYS if kind=='environment' else {'environment','testLabel','projectLabel'}
    label_values=[str(values[k]).strip().lower() for k in labels if k in values]
    designated=any(v in TEST_LABELS for v in label_values)
    return {'source_alias':alias,'endpoint_candidate_present':any(endpoint(values.get(k)) for k in urls),
        'worker_token_candidate_present':any(configured(values.get(k)) for k in tokens),
        'explicit_test_designation_present':designated,
        'test_label_classification':'known test label' if designated else 'missing or unverified',
        'values_emitted':False}

def read_small(path):
    try:
        # No symlinks/reparse or external references; inspect only named local config files.
        relative=path.absolute().relative_to(GPU.absolute())
        current=GPU
        for component in [None,*relative.parts]:
            if component is not None:current=current/component
            info=current.lstat()
            if current.is_symlink() or getattr(info,'st_file_attributes',0)&0x400:return None,'reparse_skipped'
        with path.open('rb') as stream:raw=stream.read(65537)
        if len(raw)>65536:return None,'oversize_skipped'
        if path.name.startswith('.env'):
            result={}
            for line in raw.decode('utf-8-sig','replace').splitlines():
                match=re.match(r'\s*(?:export\s+)?([A-Za-z_][A-Za-z0-9_]*)\s*=\s*(.*)',line)
                if match:result[match[1]]=match[2].strip().strip('\"\'')
            return result,'environment'
        value=json.loads(raw.decode('utf-8-sig'))
        return (value,'json') if isinstance(value,dict) else (None,'non_object_skipped')
    except FileNotFoundError:return None,'missing'
    except (OSError,ValueError):return None,'unreadable_or_invalid'
