NOW = time.time()
def valid(owner='pc-ai', now=NOW):
    return {'version':1,'owner':owner,'issuedAtUtc':datetime.fromtimestamp(now,timezone.utc).isoformat(),
            'issuedAtUnix':now,'admitUntilUnix':now+120,'policy':'CPU-only independent QINT-P retest'}
CASES = [None, [], True, {}, {'owner':'nobody'}]
for key, values in {
    'version':[True, None, '1', -1, 2],
    'admitUntilUnix':[True,None,'100',float('nan'),float('inf'),-float('inf'),10**999,0,NOW-1,NOW+3601],
    'issuedAtUnix':[True,None,'100',float('nan'),10**999,NOW-2],
    'issuedAtUtc':[None,123,'2026-11-07T08:00:00-05:00','not-a-date','2026-11-07T08:00:00'],
    'policy':[True,'x'*161],
}.items():
    for bad in values:
        row = valid(); row[key] = bad; CASES.append(row)
row = valid(); row['command'] = 'bad'; CASES.append(row)
row = valid(now=NOW+300); CASES.append(row)  # Remains future-invalid throughout native process startup matrix.

class Matrix(unittest.TestCase):
    def test_shape_finite_window_and_reader_agreement(self):
        guard = candidate()
        for row in CASES:
            with self.subTest(row=str(row)[:120]):
                with self.assertRaises((ValueError,TypeError)): guard.validate_slot(row,NOW)
                with self.assertRaises(ValueError): slot(row,NOW)
    def test_valid_legacy_and_v1_window(self):
        guard = candidate()
        for row in [valid(), {key:value for key,value in valid().items() if key not in {'version','issuedAtUnix'}},
                    {**valid(), 'admitUntilUnix':NOW+3600}]:
            self.assertEqual(guard.validate_slot(row,NOW),'pc-ai'); self.assertEqual(slot(row,NOW),row['admitUntilUnix'])
    def test_invalid_clock(self):
        guard = candidate()
        for now in [True,None,'100',float('nan'),float('inf'),-float('inf'),10**999,0]:
            with self.subTest(clock=str(now)[:30]), self.assertRaises(ValueError): guard.validate_slot(valid(),now)

class NativeCandidate(old.NativeCPUTests):
    def setUp(self):
        super().setUp()
        (self.root/'cpu-test-slot.json').write_text(json.dumps(valid(now=time.time())))
    def command(self,*args,**kw):
        result = super().command(*args,**kw)
        result[1] = str(ROOT/'candidate_driver.py')
        return result
    def test_slot_refused_before_child_launch(self):
        wrong = valid('pc-3d',now=time.time())
        for row, expected in [(wrong,75), *[(row,70) for row in CASES],
                              ('{"owner":"pc-ai","owner":"pc-ai"}',70),
                              ('{"owner":"pc-ai","admitUntilUnix":NaN}',70)]:
            (self.root/'cpu-test-slot.json').write_text(row if isinstance(row,str) else json.dumps(row))
            result = subprocess.run(self.command('normal'),env=self.env,capture_output=True,text=True,timeout=8)
            with self.subTest(row=str(row)[:100]):
                self.assertEqual(result.returncode,expected,result.stderr)
                self.assertFalse((self.root/'child.json').exists())
        self.event(invalid_cases=len(CASES)+2,wrong_owner_exit=75,no_child_launched=True)
