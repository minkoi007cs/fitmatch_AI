def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()

def safe_env():
    values={key:os.environ[key] for key in ['SystemRoot','WINDIR','PATH','TEMP','TMP','COMSPEC','PATHEXT'] if key in os.environ}
    values.update(PYTHONIOENCODING='utf-8',PYTHONUNBUFFERED='1',NO_PROXY='127.0.0.1,localhost',no_proxy='127.0.0.1,localhost')
    return values

def peer_acquires(path):
    code=('import msvcrt,sys;f=open(sys.argv[1],"r+");f.seek(0)\n'
          'try:msvcrt.locking(f.fileno(),msvcrt.LK_NBLCK,1)\n'
          'except OSError:sys.exit(3)\n'
          'f.seek(0);msvcrt.locking(f.fileno(),msvcrt.LK_UNLCK,1)')
    return subprocess.run([psutil.Process().exe(),'-c',code,str(path)],env=safe_env(),
                          creationflags=0x08000000,timeout=5).returncode==0

class NativeWitness:
    """Retain SYNCHRONIZE handles to adapter interpreters inside our owned outer Job Object."""
    def __init__(self,audit_dir,outer):
        self.audit_dir=audit_dir;self.outer=outer;self.handles={};self.stopped=threading.Event()
        self.mutex=threading.RLock();self.events=[]
        self.k=ctypes.WinDLL('kernel32',use_last_error=True)
        self.k.OpenProcess.restype=wintypes.HANDLE
        self.k.OpenProcess.argtypes=(wintypes.DWORD,wintypes.BOOL,wintypes.DWORD)
        self.k.IsProcessInJob.argtypes=(wintypes.HANDLE,wintypes.HANDLE,ctypes.POINTER(wintypes.BOOL))
        self.k.WaitForSingleObject.argtypes=(wintypes.HANDLE,wintypes.DWORD)
        self.k.WaitForSingleObject.restype=wintypes.DWORD
        self.k.CloseHandle.argtypes=(wintypes.HANDLE,)
        self.thread=threading.Thread(target=self.loop,daemon=True);self.thread.start()
    def read_events(self):
        events=[]
        for path in self.audit_dir.glob('*.jsonl'):
            for line in path.read_text().splitlines():
                try:events.append(json.loads(line))
                except ValueError:pass
        return events
    def scan(self):
        with self.mutex:
            self.events=self.read_events()
            for event in self.events:
                pid=event['pid']
                if event['script'] not in ['worker_cli.py','mock_qwen_adapter.py'] or pid in self.handles:continue
                handle=self.k.OpenProcess(0x00100000|0x1000,False,pid)
                if not handle:continue
                inside=wintypes.BOOL()
                if self.k.IsProcessInJob(handle,self.outer.job.handle,ctypes.byref(inside)) and inside.value:
                    self.handles[pid]=handle
                else:self.k.CloseHandle(handle)
    def loop(self):
        while not self.stopped.wait(.002):self.scan()
    def at_report(self):
        self.scan()
        with self.mutex:
            adapters={row['pid'] for row in self.events if row['phase']=='start' and row['script'] in ['worker_cli.py','mock_qwen_adapter.py']}
            assert adapters,'real adapter interpreter was not witnessed'
            assert adapters<=set(self.handles),'adapter native handle missed; cannot prove signal-before-release'
            signaled=all(self.k.WaitForSingleObject(self.handles[pid],0)==0 for pid in adapters)
            assert signaled,'worker reported before adapter process object was signaled'
            assert all(not row.get('workerTokenPresent',False) for row in self.events if row['script'] in ['worker_cli.py','mock_qwen_adapter.py'])
            assert not any(row['phase']=='FORBIDDEN_TORCH_IMPORT' or row.get('torchImported') for row in self.events)
            return {'adapterInterpreterCount':len(adapters),'nativeAdapterHandlesAllSignaledAtHTTPReport':signaled,
                    'adapterCredentialsRemoved':True,'torchImportAttempted':False}
    def close(self):
        self.stopped.set();self.thread.join(timeout=2)
        with self.mutex:
            for handle in self.handles.values():self.k.CloseHandle(handle)

def coordination_fingerprint():
    root=GPU/'coordination'
    paths=['gpu-coordinator.json','gpu0-slot.json','gpu0-state.json']
    result={name:sha(root/name) for name in paths}
    stat=(root/'gpu0.lock').stat()
    result['lockMetadata']=[stat.st_dev,stat.st_ino,stat.st_size,stat.st_mtime_ns]
    result['guardSHA256']=sha(GPU/'fitmatch_AI/workers/windows/gpu_guard.py')
    result['operatorApprovalPresent']=(root/'qint-worker-test-approval.json').exists()
    return result
