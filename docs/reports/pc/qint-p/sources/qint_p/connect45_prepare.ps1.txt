function Protect-ConnectPath([string]$Path, [bool]$Directory) {
    $connectPresentAcl = Get-Acl -LiteralPath $Path
    $connectPresentRules = @($connectPresentAcl.GetAccessRules($true,$true,[Security.Principal.SecurityIdentifier]))
    if ($connectPresentAcl.AreAccessRulesProtected -and $connectPresentRules.Count -eq 1 -and
        $connectPresentRules[0].IdentityReference.Value -eq $connectSid.Value -and
        $connectPresentRules[0].AccessControlType -eq [Security.AccessControl.AccessControlType]::Allow -and
        $connectPresentRules[0].FileSystemRights -eq [Security.AccessControl.FileSystemRights]::FullControl -and
        $connectPresentAcl.GetOwner([Security.Principal.SecurityIdentifier]).Value -eq $connectSid.Value) { return }
    if ($Directory) {
        $connectAcl = [Security.AccessControl.DirectorySecurity]::new()
        $connectRule = [Security.AccessControl.FileSystemAccessRule]::new($connectSid,
            [Security.AccessControl.FileSystemRights]::FullControl,
            ([Security.AccessControl.InheritanceFlags]::ContainerInherit -bor [Security.AccessControl.InheritanceFlags]::ObjectInherit),
            [Security.AccessControl.PropagationFlags]::None, [Security.AccessControl.AccessControlType]::Allow)
    } else {
        $connectAcl = [Security.AccessControl.FileSecurity]::new()
        $connectRule = [Security.AccessControl.FileSystemAccessRule]::new($connectSid,
            [Security.AccessControl.FileSystemRights]::FullControl,[Security.AccessControl.AccessControlType]::Allow)
    }
    $connectAcl.SetOwner($connectSid)
    $connectAcl.SetAccessRuleProtection($true,$false)
    $connectAcl.AddAccessRule($connectRule)
    Set-Acl -LiteralPath $Path -AclObject $connectAcl
    $connectVerified = Get-Acl -LiteralPath $Path
    $connectRules = @($connectVerified.GetAccessRules($true,$true,[Security.Principal.SecurityIdentifier]))
    if (-not $connectVerified.AreAccessRulesProtected -or $connectRules.Count -ne 1 -or
        $connectRules[0].IdentityReference.Value -ne $connectSid.Value -or
        $connectRules[0].AccessControlType -ne [Security.AccessControl.AccessControlType]::Allow -or
        $connectVerified.GetOwner([Security.Principal.SecurityIdentifier]).Value -ne $connectSid.Value) {
        throw 'Owner-only ACL verification failed; token must not be used'
    }
}
function Connect-Sha([byte[]]$Bytes) {
    $connectHasher = [Security.Cryptography.SHA256]::Create()
    try { return ([BitConverter]::ToString($connectHasher.ComputeHash($Bytes))).Replace('-','').ToLowerInvariant() }
    finally { $connectHasher.Dispose() }
}
