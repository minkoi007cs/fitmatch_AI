"""Single final CPU evidence suite, with actual Windows process cases and memory sampling."""
import json
from pathlib import Path
import threading
import time
import unittest
import psutil
import test_control,test_envelope,test_retention,test_metrics,test_corpus,test_native_cpu

ROOT=Path(__file__).resolve().parent;RAW=ROOT.parents[1]/'reports'/'ai'/'quality03'
samples=[];stop=threading.Event();started=time.perf_counter()
def observe():
    while not stop.is_set():
        try:
            process=psutil.Process();tree=[process,*process.children(recursive=True)]
            rss=sum(p.memory_info().rss for p in tree if p.is_running())
            samples.append({'seconds':time.perf_counter()-started,'process_tree_rss':rss,'host_available':psutil.virtual_memory().available})
        except psutil.Error:pass
        stop.wait(.05)
monitor=threading.Thread(target=observe,daemon=True);monitor.start()
suite=unittest.TestSuite()
for module in [test_control,test_envelope,test_retention,test_metrics,test_corpus,test_native_cpu]:
    suite.addTests(unittest.defaultTestLoader.loadTestsFromModule(module))
result=unittest.TextTestRunner(verbosity=2).run(suite)
stop.set();monitor.join(timeout=1)
output={'session':'PC-AI-QUALITY-20261106-03','scope':'CPU fixtures, logical controller, envelope/retention proposal and isolated native Windows process tests',
        'tests_run':result.testsRun,'failures':len(result.failures),'errors':len(result.errors),'skipped':[reason for _,reason in result.skipped],
        'passed':result.testsRun-len(result.failures)-len(result.errors)-len(result.skipped),'exit_code':0 if result.wasSuccessful() else 1,
        'seconds':round(time.perf_counter()-started,3),'gpu_model_loaded':False,'real_machine_memory_exhaustion':False,
        'native_events':test_native_cpu.EVENTS,'peak_sampled_process_tree_rss_bytes':max(x['process_tree_rss'] for x in samples),
        'min_sampled_host_available_bytes':min(x['host_available'] for x in samples),
        'process_peak_working_set_bytes':psutil.Process().memory_info().peak_wset,'memory_samples':samples}
(RAW/'cpu-suite.json').write_text(json.dumps(output,indent=2)+'\n',encoding='utf-8')
raise SystemExit(output['exit_code'])
